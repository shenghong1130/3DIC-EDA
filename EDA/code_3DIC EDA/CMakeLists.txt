cmake_minimum_required(VERSION 3.16)
project(gr3dic VERSION 0.1 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_executable(router src/main.cpp src/parser.cpp src/router.cpp)
target_include_directories(router PRIVATE include)
add_executable(router_v3_2 EXCLUDE_FROM_ALL src/main.cpp src/parser.cpp src/router.cpp)
target_include_directories(router_v3_2 PRIVATE include)
target_compile_definitions(router_v3_2 PRIVATE GR_V3_FINAL GR_V3_2)
add_executable(router_v3_3 EXCLUDE_FROM_ALL src/main.cpp src/parser.cpp src/router.cpp)
target_include_directories(router_v3_3 PRIVATE include)
target_compile_definitions(router_v3_3 PRIVATE GR_V3_FINAL GR_V3_3)
if(WIN32)
  target_link_libraries(router_v3_3 PRIVATE shell32)
endif()
if(WIN32)
  target_link_libraries(router_v3_2 PRIVATE shell32)
endif()
if(MSVC)
  target_compile_options(router PRIVATE /W4 /utf-8)
else()
  target_compile_options(router PRIVATE -Wall -Wextra -Wpedantic)
endif()
if(WIN32)
  target_link_libraries(router PRIVATE shell32)
endif()
include(CTest)
if(BUILD_TESTING)
  add_executable(core_tests tests/core_tests.cpp src/parser.cpp src/router.cpp)
  target_include_directories(core_tests PRIVATE include)
  add_test(NAME core COMMAND core_tests)
endif()
